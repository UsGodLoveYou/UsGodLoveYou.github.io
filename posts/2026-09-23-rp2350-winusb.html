RP2350 WINUSB 自动装驱动 — 完整实战方案

RP2350 Pico 2 WINUSB TinyUSB USB Embedded
PICO_STDIO_USB_RESET_INTERFACE_SUPPORT_MS_OS_20_DESCRIPTOR=1 PICO_STDIO_USB_RESET_INTERFACE_MS_OS_20_DESCRIPTOR_ITF=2 PICO_STDIO_USB_RESET_INTERFACE_SUPPORT_RESET_TO_BOOTSEL=1 PICO_STDIO_USB_RESET_BOOTSEL_INTERFACE_DISABLE_MASK=0 )

然后设备接口：

Interface 0: CDC Control
Interface 1: CDC Data
Interface 2: Vendor (Reset / WINUSB)  ← Windows 自动装 WinUSB.sys

5. 完整代码（基于 pico-examples dev_multi_cdc）

5.1 CMakeLists.txt

cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(rp2350_winusb C CXX ASM)
pico_sdk_init()

add_executable(rp2350_winusb
    main.c
    usb_descriptors.c
)

target_include_directories(rp2350_winusb PRIVATE
    ${CMAKE_CURRENT_LIST_DIR}
    ${pico-sdk-src}/src/rp2_common/pico_stdio_usb/include
)

target_link_libraries(rp2350_winusb PRIVATE
    pico_stdlib
    pico_unique_id
    tinyusb_device
    tinyusb_board
)

pico_enable_stdio_usb(rp2350_winusb 1)
target_compile_definitions(rp2350_winusb PRIVATE
    PICO_STDIO_USB_ENABLE_RESET_VIA_VENDOR_INTERFACE=1
    PICO_STDIO_USB_RESET_INTERFACE_SUPPORT_MS_OS_20_DESCRIPTOR=1
    PICO_STDIO_USB_RESET_INTERFACE_MS_OS_20_DESCRIPTOR_ITF=2
    PICO_STDIO_USB_RESET_INTERFACE_SUPPORT_RESET_TO_BOOTSEL=1
    PICO_STDIO_USB_RESET_BOOTSEL_INTERFACE_DISABLE_MASK=0
)

pico_add_extra_outputs(rp2350_winusb)

5.2 usb_descriptors.c（改自 dev_multi_cdc）

关键修改：bcdUSB = 0x0210，VID/PID 用 Raspberry Pi：

#define CDC_EXAMPLE_VID     0x2E8A  // Raspberry Pi VID
#define CDC_EXAMPLE_PID     0x000A  // LevelekLs WINUSB PID
#define CDC_EXAMPLE_BCD     0x0210  // USB 2.1 (Windows reads MSOS 2.0)

其余 CDC 接口配置不变（参考 dev_multi_cdc）。

6. 实际效果

烧录后，Windows 设备管理器：

USB 控制器
└── USB Composite Device
    ├── USB Serial Device (COM3)            ← CDC, Windows 自动驱动
    └── LevelekLs WINUSB Device             ← WINUSB, WinUSB.sys 自动装

无黄色感叹号，无 Zadig，无 .inf 文件。

PowerShell 验证：

PS> Get-PnpDevice -InstanceId "USB\VID_2E8A*"

------     -----           ------------
OK         USB Serial         USB Serial Device (COM3)
OK         USB               LevelekLs WINUSB Device

7. 主机端访问

7.1 Python libusb

import usb.core
import usb.util

dev = usb.core.find(idVendor=0x2E8A, idProduct=0x000A)
if dev is None:
    raise ValueError("Device not found")

# claim WINUSB interface (interface 2)
dev.set_configuration()
ep_out = dev[0][(2, 0)][0]  # WINUSB bulk OUT
ep_in  = dev[0][(2, 0)][1]  # WINUSB bulk IN

# send/receive
ep_out.write(b"Hello WINUSB\n")
data = ep_in.read(64)

7.2 C++ WinUSB API

#include <windows.h>
#include <winusb.h>

HANDLE hDevice = CreateFile(L"\\\\?\\YourDevice#...", ...);
WINUSB_INTERFACE_HANDLE hInterface;
WinUsb_Initialize(hDevice, &hInterface);
// ...bulk transfer...

8. 调试踩坑总结

问题 原因 解决
bcdUSB=0x0200 不读 BOS Windows 1.x 不强制读 改 0x0210
GUID 写错 用错 MSOS 2.0 GUID 用 {D8DD60DF-4589-4CC7-9CD2-659D9E648A9F}
bDevCapabilityType=0x00 应为 0x05 PLATFORM 改 0x05
wIndex 不检查 必须 = 0x07 加检查
bmRequestType mask 错 0xC0 & 0x60 != 0x40 精确比较 0xC0
自己写描述符不全 缺 Registry Property 用 SDK reset_interface
dev_lowlevel 不支持分包 限制 64 字节 用完整 TinyUSB stack

10. 结论

RP2350 + Pico SDK 实现 WINUSB 自动装驱动 的关键：

  1. ✅ 使用 SDK 内置 reset_interface（自动提供 MS OS 2.0 描述符）
  2. ✅ bcdUSB = 0x0210
  3. ✅ 启用 PICO_STDIO_USB_RESET_INTERFACE_SUPPORT_MS_OS_20_DESCRIPTOR=1
  4. ✅ VID = 0x2E8A + PID 唯一

整个方案不需要自己写 166 字节 MS OS 2.0 描述符——SDK 已经处理好了。
